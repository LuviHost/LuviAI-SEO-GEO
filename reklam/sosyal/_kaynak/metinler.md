```bash
cd reklam/sosyal

"$CHROME" --headless=new --force-device-scale-factor=2 \
  --screenshot="$PWD/ranksup-case-linkedin.png" --window-size=1200,1200 \
  "file://$PWD/case-linkedin.html"

"$CHROME" --headless=new --force-device-scale-factor=2 \
  --screenshot="$PWD/ranksup-case-x.png" --window-size=1600,900 \
  "file://$PWD/case-x.html"
```
